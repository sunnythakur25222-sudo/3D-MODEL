import time
import math
import random

# Hardware Pin Configurations & Addresses
ESP32_CAM_PIN     = 34   # Horizon hazard detection optical camera feed
SHARP_DUST_PIN    = 35   # Sharp GP2Y1014AU0F dust density sensor
BH1750_I2C_ADDR   = 0x23 # BH1750 ambient light lux level sensor

# L298N Dual H-Bridge Motor Driver Control Pins
L298N_ENA = 25  # PWM Speed Control Enable
L298N_IN1 = 26  # Piston Actuator Extend Signal
L298N_IN2 = 27  # Piston Actuator Retract Signal

# Operational System Threshold Constants
DUST_THRESHOLD_MG_M3 = 1.20   # Hazard trigger (> 1.2 mg/m³)
LIGHT_MIN_THRESHOLD_LUX = 250  # Solar dimness trigger (< 250 Lux)

class MarsRoverShieldController:
    def __init__(self):
        self.shield_deployed = False
        self.actuator_position = 0.0  # 0.0 = Retracted, 1.0 = Extended (100%)
        print("[INIT] Mars Rover Autonomous Shield Defence Initialized.")

    def read_esp32_cam_feed(self):
        """Simulates visual optical scanning from front mast camera."""
        return {"fps": 60, "optical_hazard": random.choice([False, False, True])}

    def read_sharp_gp2y1014(self, storm_simulated=False):
        """Reads dust particle density in mg/m3 from Sharp sensor."""
        if storm_simulated:
            return round(random.uniform(2.1, 4.5), 2)
        return round(random.uniform(0.01, 0.25), 2)

    def read_bh1750_light_sensor(self, storm_simulated=False):
        """Reads ambient illumination lux levels via I2C."""
        if storm_simulated:
            return random.randint(80, 220)
        return random.randint(750, 980)

    def set_l298n_actuators(self, direction, pwm_speed=255):
        """Drives linear actuator motors via L298N dual H-bridge output."""
        if direction == "EXTEND":
            # HIGH on IN1, LOW on IN2 -> Piston Deploy Canopy Dome
            print(f"[ACTUATOR] Driving L298N IN1=HIGH, IN2=LOW | PWM={pwm_speed}")
            self.shield_deployed = True
            self.actuator_position = 1.0
        elif direction == "RETRACT":
            # LOW on IN1, HIGH on IN2 -> Piston Stow Canopy
            print(f"[ACTUATOR] Driving L298N IN1=LOW, IN2=HIGH | PWM={pwm_speed}")
            self.shield_deployed = False
            self.actuator_position = 0.0

    def evaluate_defence_logic(self, storm_active=False):
        """Core decision loop executing sensor 'AND' trigger logic."""
        dust_val = self.read_sharp_gp2y1014(storm_active)
        light_val = self.read_bh1750_light_sensor(storm_active)
        cam_feed = self.read_esp32_cam_feed()

        print(f"[TELEMETRY] Dust: {dust_val} mg/m³ | Light: {light_val} Lux")

        # Logic: Trigger shield if dust exceeds threshold OR solar drop detected
        if dust_val > DUST_THRESHOLD_MG_M3 or light_val < LIGHT_MIN_THRESHOLD_LUX:
            if not self.shield_deployed:
                print("[ALERT] Dust storm hazard confirmed! Deploying full shield canopy...")
                self.set_l298n_actuators("EXTEND")
        else:
            if self.shield_deployed:
                print("[INFO] Atmosphere clear. Retracting umbrella shield...")
                self.set_l298n_actuators("RETRACT")

        return {
            "dust": dust_val,
            "light": light_val,
            "shield": "DEPLOYED (100%)" if self.shield_deployed else "STOWED (0%)"
        }

# Instantiate and run execution cycle
